{{- /* Markdown tables. Adds data-label (the column header) to each cell so tables can stack into cards on phones, and turns verification labels (VERIFIED, VENDOR-REPORTED, ...) into coloured badges. */ -}} {{- $headers := slice -}} {{- range first 1 .THead }}{{ range . }}{{ $headers = $headers | append (.Text | plainify | htmlUnescape | strings.TrimSpace) }}{{ end }}{{ end -}} {{- $badges := dict "PARTIALLY VERIFIED" "partial" "VENDOR-REPORTED" "vendor" "UNVERIFIED" "unverified" "VERIFIED" "verified" -}} {{- range .THead }} {{- range . }} {{ .Text }} {{- end }} {{- end }} {{- range .TBody }} {{- range $i, $cell := . }} {{- $html := $cell.Text | string -}} {{- with index (findRE `^(PARTIALLY VERIFIED|VENDOR-REPORTED|UNVERIFIED|VERIFIED)\b` (strings.TrimSpace $html) 1) 0 -}} {{- $html = replaceRE (printf `^\s*%s` .) (printf `%s` (index $badges .) .) $html 1 -}} {{- end }} {{ $html | safeHTML }} {{- end }} {{- end }}
